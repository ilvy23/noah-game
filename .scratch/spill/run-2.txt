def check_ending(s):
    """Return an ending id if the run has to end now, else None.

    The one forced ending: if your head goes completely empty, the valley
    does not let you keep going. You simply *become* the next line.

    A second one: if you have paid the cost to the archivist and then
    walked away from the schloss, the valley does not let you keep the
    name. it takes it back. this is the only ending that can happen
    *outside* the cellar.
    """
    if s.sanity <= 0:
        return "folk"
    if s.has("archivist_paid") and s.location != "schloss" and not s.has("made_choice"):
        return "paid"
    return None

def play_location(loc_id: str) -> bool:
    """Start music for a location id.

    - Unknown location id: keep the current track (no new player started).
    - Same track already playing: do NOT restart it.
    - Different track: stop the old player first, then start the new one.
    Returns True if a player is running for this location afterwards.
    """
    track = LOCATION_TRACKS.get(loc_id)
    if track is None:
        return _current_proc is not None and _current_proc.poll() is None
    if track == _current_track and _current_proc is not None \
            and _current_proc.poll() is None:
        return True
    stop()
    return _start_player(track)


def start(loc_id: str) -> bool:
    """Start music for the opening location (used at game start)."""
    return play_location(loc_id)


def is_playing() -> bool:
    return _current_proc is not None and _current_proc.poll() is None


def current_track() -> str:
    return _current_track

def talk_npc(s, who):
    context.record(s, "talk", who=who, place=s.location)
    tier = talk_tier(s)
    # --- new dialogue trees (dialogue.py) take priority when available ---
    if who == "connor":
        open_text = dialogue.CONNOR_OPEN.get(tier, dialogue.CONNOR_OPEN["low"])
        dialogue.run_tree(s, "connor", dialogue.CONNOR_TREE, open_text)
        s.flag("met_connor")
    elif who == "mika":
        open_text = dialogue.MIKA_OPEN.get(tier, dialogue.MIKA_OPEN["low"])
        dialogue.run_tree(s, "mika", dialogue.MIKA_TREE, open_text)
        s.flag("met_mika")
    elif who == "archivist":
        open_text = dialogue.ARCHIVIST_OPEN.get(tier, dialogue.ARCHIVIST_OPEN["low"])
        dialogue.run_tree(s, "archivist", dialogue.ARCHIVIST_TREE, open_text)
        s.flag("met_archivist")
    elif who == "noah":
        # Noah uses the raw sanity tier (all 4 tiers are reachable)
        raw_tier = s.sanity_tier
        open_text = dialogue.NOAH_OPEN.get(raw_tier, dialogue.NOAH_OPEN["low"])
        dialogue.run_tree(s, "noah", dialogue.NOAH_TREE, open_text)
        s.flag("met_noah")
        # Sanity restore with cooldown: +8 sanity once per day
        if s.day != s._noah_rest_day:
            s._noah_rest_day = s.day
            s.sanity_change(8, "sitting with Noah")
            render("")
            render("  the tea is warm. the bench is the right size for two.")
            render("  you sit. the valley is patient. you are not.")
    else:
        # fallback to the old flat lines (only reachable for unknown NPC ids)
        render("")
        render("  no one is here. the light is, though.")

    # chapter progression after any conversation
    msg = story.advance_chapter(s)
    if msg:
        render("")
        render(msg)

    achievements.check_all(s)

def examine_item(s, iid):
    item = locations.ITEMS[iid]
    render(item["name"])
    rule()
    render(item["text"])
    s.take(iid)
    s.sanity_change(-2, "examined " + item["name"])

    # special beats
    if iid == "letter":
        s.flag("read_letter")
    if iid == "ledger":
        if not s.has("read_ledger"):
            s.flag("read_ledger")
            s.sanity_change(-10, "read the ledger")
        else:
            render("")
            render("The last line is different. It is your name, but written")
            render("*from the other side of the page*.")
            s.flag("read_ledger_twice")
            s.sanity_change(-12, "read the ledger again")
    elif iid == "notebook":
        s.flag("met_connor")
    elif iid == "ritual_text":
        s.flag("has_instruction")
    elif iid == "key":
        s.flag("has_key")

    # chapter progression: check if the current chapter's objective is met
    msg = story.advance_chapter(s)
    if msg:
        render("")
        render(msg)

    achievements.check_all(s)
    pause()

def chapter_line(s, loc_id: str) -> str:
    """Return a short chapter-specific line for the given location.

    The line picks up the player's last interaction (via context.ref) so
    the description is a continuation of what just happened, not a
    standalone line. Returns "" if there is no line for the current
    chapter."""
    ch = story.current_chapter(s)
    lines = CHAPTER_LINES.get(loc_id, {})
    base = lines.get(ch, "")
    if not base:
        return ""
    prev = context.ref(s)
    if prev:
        return f"  ({prev} — and now this.)\n{base}"
    return base
