def current_chapter(s) -> int:
    """Return the first chapter whose done_flag is not set (1..5)."""
    for i in range(1, 6):
        if not s.has(CHAPTERS[str(i)]["done_flag"]):
            return i
    return 5


def chapter_objective(s) -> str:
    """The objective text for the current chapter."""
    ch = current_chapter(s)
    return CHAPTERS[str(ch)]["objective"]


def chapter_title(s) -> str:
    """The title of the current chapter."""
    ch = current_chapter(s)
    return CHAPTERS[str(ch)]["title"]


def chapter_done(s, ch: int) -> bool:
    """True if the given chapter is complete."""
    return s.has(CHAPTERS[str(ch)]["done_flag"])


def chapter_condition_met(s, ch: int) -> bool:
    """True if the current chapter's objective is satisfied.

    The conditions use flags that already exist in the game:
      ch1: read the letter (examine letter) + met someone (connor/mika/archivist)
      ch2: talked to Connor (connor_told or connor_notebook_read)
      ch3: talked to Mika (asked_mika_painting or mika_why)
      ch4: asked the archivist about the cost + read the ledger
      ch5: has key + ritual text + ledger read (ritual_unlocked)
    """
    if ch == 1:
        return (s.has("read_letter")
                and (s.has("met_connor") or s.has("met_mika")
                     or s.has("met_archivist")))
    if ch == 2:
        return (s.has("connor_told") or s.has("connor_notebook_read")
                or s.has("connor_dates_asked"))
    if ch == 3:
        return (s.has("asked_mika_painting") or s.has("mika_why")
                or s.has("asked_mika_valley"))
    if ch == 4:
        return (s.has("asked_archivist_cost") and s.has("read_ledger"))
    if ch == 5:
        return ritual_unlocked(s)
    return False

def advance_chapter(s) -> str:
    """Check if the current chapter's condition is met. If so, set the
    done_flag, advance s.chapter, and return a short in-world message.
    Returns "" if no chapter advanced."""
    ch = current_chapter(s)
    if ch >= 5:
        return ""
    if not chapter_condition_met(s, ch):
        return ""
    s.flag(CHAPTERS[str(ch)]["done_flag"])
    s.chapter = ch + 1
    next_ch = CHAPTERS[str(ch + 1)]
    # Pick up what the player just did, so the chapter turn is not a
    # standalone line but a consequence of the last interaction.
    prev = context.ref(s)
    if prev:
        return (f"\n— the week turns. —\n"
                f"  {prev}. the valley notes it.\n"
                f"  chapter {ch + 1}: {next_ch['title']}\n"
                f"  objective: {next_ch['objective']}")
    return (f"\n— the week turns. —\n"
            f"  chapter {ch + 1}: {next_ch['title']}\n"
            f"  objective: {next_ch['objective']}")

def show_journal(s):
    """The journal command: everything you have learned and decided.

    Displays four categories built from the current game state:
      · people met   — from s.journal entries with kind='met'
      · places visited — from s.visited
      · items found    — from s.inventory
      · facts learned  — from s.journal entries with kind='learned'
    """
    render("")
    rule(60)
    render("  JOURNAL — what you have learned and decided")
    rule(60)
    render("")

    # --- People met ---
    render("  PEOPLE MET")
    met = [e["text"] for e in s.journal if e.get("kind") == "met"]
    if met:
        for name in met:
            render("    · " + name)
    else:
        render("    (no one yet)")
    render("")

    # --- Places visited ---
    render("  PLACES VISITED")
    if s.visited:
        for loc_id in s.visited:
            loc = locations.LOCATIONS.get(loc_id, {})
            name = loc.get("name", loc_id)
            render("    · " + name)
    else:
        render("    (nowhere yet)")
    render("")

    # --- Items found ---
    render("  ITEMS FOUND")
    if s.inventory:
        for iid in s.inventory:
            item = locations.ITEMS.get(iid, {})
            name = item.get("name", iid)
            render("    · " + name)
    else:
        render("    (nothing yet)")
    render("")

    # --- Facts learned ---
    render("  FACTS LEARNED")
    learned = [e["text"] for e in s.journal if e.get("kind") == "learned"]
    if learned:
        for fact in learned:
            render("    · " + fact)
    else:
        render("    (nothing yet)")
    render("")

    # --- Other journal entries (decided, death) ---
    other = [e for e in s.journal if e.get("kind") in ("decided", "death")]
    if other:
        render("  OTHER")
        for e in other:
            kind = e.get("kind", "")
            text = e.get("text", "")
            if kind == "decided":
                render("    · decided   " + text)
            elif kind == "death":
                render("    · died      " + text)
            else:
                render("    · " + text)
        render("")

    render("  chapter  " + str(story.current_chapter(s)) + " — " + story.chapter_title(s))
    render("  objective  " + story.chapter_objective(s))
    render("")
    render("  flags set: " + (", ".join(sorted(s.flags)) if s.flags else "(none)"))
    render("  trust " + str(s.trust) + "   fear " + str(s.fear))
    render("")
    # topics you can still ask about, per NPC
    for who in ("connor", "mika", "archivist"):
        topics = sorted(dialogue.ASK_TOPICS.get(who, {}).keys())
        if topics:
            render("  ask " + who + ": " + ", ".join(topics))
    render("")
